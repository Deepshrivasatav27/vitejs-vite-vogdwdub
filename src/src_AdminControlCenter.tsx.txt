import { useEffect, useMemo, useState } from 'react'
import { supabase } from './supabase'
import './AdminControlCenter.css'

type Tab = 'overview' | 'leads' | 'integrations' | 'plans' | 'features' | 'pages' | 'audit'
type LeadRow = { id:string; customer_name:string; mobile:string|null; source:string|null; status:string|null; assigned_to:string|null; created_at:string; company_id:string }
type Profile = { id:string; full_name:string|null; role:string|null; company_id:string|null }
type Plan = { name:string; price:string; duration:string; description:string; enabled:boolean }
type PageContent = { title:string; body:string; published:boolean }
const defaultPlans:Plan[]=[{name:'Free',price:'0',duration:'30 days',description:'Starter access for small teams',enabled:true},{name:'Pro',price:'999',duration:'Monthly',description:'Full CRM for growing teams',enabled:true},{name:'Business',price:'2499',duration:'Monthly',description:'Advanced controls and reporting',enabled:true},{name:'Enterprise',price:'Custom',duration:'Yearly',description:'Custom limits and support',enabled:true}]
const defaultFeatures=['Lead Management','Lead Assignment & Transfer','Import / Export','Calling Center','WhatsApp Center','Site Visit Management','Analytics & Reports','API Integrations','Workflow Automation','Custom Fields','Team Management','Audit Logs','Payment & Subscription','Privacy Policy Editor','About Page Editor']
export default function AdminControlCenter(){
 const [tab,setTab]=useState<Tab>('overview'); const [leads,setLeads]=useState<LeadRow[]>([]); const [profiles,setProfiles]=useState<Profile[]>([]); const [companies,setCompanies]=useState<{id:string;name:string}[]>([]); const [selectedCompany,setSelectedCompany]=useState(''); const [loading,setLoading]=useState(false); const [notice,setNotice]=useState(''); const [search,setSearch]=useState(''); const [plans,setPlans]=useState<Plan[]>(()=>JSON.parse(localStorage.getItem('cvx-plans-draft')||'null')||defaultPlans); const [pages,setPages]=useState<Record<string,PageContent>>(()=>JSON.parse(localStorage.getItem('cvx-pages-draft')||'null')||{about:{title:'About CODEVEX',body:'CODEVEX is a modern customer relationship management platform.',published:false},privacy:{title:'Privacy Policy',body:'Add your privacy policy here.',published:false}}); const [apiProvider,setApiProvider]=useState('99acres'); const [apiUrl,setApiUrl]=useState(''); const [apiKey,setApiKey]=useState(''); const [apiEnabled,setApiEnabled]=useState(false); const [qr,setQr]=useState(''); const [upi,setUpi]=useState('');
 useEffect(()=>{localStorage.setItem('cvx-plans-draft',JSON.stringify(plans))},[plans]); useEffect(()=>{localStorage.setItem('cvx-pages-draft',JSON.stringify(pages))},[pages]);
 async function load(){setLoading(true);setNotice('');try{const [c,p]=await Promise.all([supabase.from('companies').select('id,name').order('name'),supabase.from('profiles').select('id,full_name,role,company_id')]);if(c.error)throw c.error;if(p.error)throw p.error;setCompanies(c.data||[]);setProfiles(p.data||[]);if(!selectedCompany&&c.data?.[0])setSelectedCompany(c.data[0].id);const q=await supabase.from('leads').select('id,customer_name,mobile,source,status,assigned_to,created_at,company_id').order('created_at',{ascending:false}).limit(300);if(q.error)throw q.error;setLeads(q.data||[])}catch(e:any){setNotice(e.message||'Could not load data')}finally{setLoading(false)}}
 useEffect(()=>{void load()},[])
 const agents=useMemo(()=>profiles.filter(p=>p.company_id===selectedCompany),[profiles,selectedCompany]); const visible=leads.filter(l=>(!selectedCompany||l.company_id===selectedCompany)&&`${l.customer_name} ${l.mobile||''} ${l.source||''}`.toLowerCase().includes(search.toLowerCase()));
 async function assign(lead:LeadRow,userId:string){if(!userId)return;const {error}=await supabase.from('leads').update({assigned_to:userId}).eq('id',lead.id).eq('company_id',lead.company_id);if(error){setNotice(error.message);return}setNotice(`Assignment updated for ${lead.customer_name}.`);void load()}
 const nav:[Tab,string,string][]=[['overview','Overview','◫'],['leads','Lead Control','♙'],['integrations','API & Integrations','⌘'],['plans','Plans & Payments','₹'],['features','Feature Manager','⚙'],['pages','Public Pages','▤'],['audit','Audit & Activity','◷']]
 return <div className="cvx-admin"><div className="cvx-admin-head"><div><div className="cvx-kicker">CODEVEX · PLATFORM CONTROL</div><h1>Super Admin Control Center</h1><p>Centralized frontend controls for platform operations.</p></div><button onClick={()=>void load()} className="cvx-btn">↻ Refresh data</button></div>
 <div className="cvx-tabs">{nav.map(([id,label,icon])=><button key={id} className={tab===id?'active':''} onClick={()=>setTab(id)}>{icon} {label}</button>)}</div>{notice&&<div className="cvx-notice">{notice}</div>}
 {tab==='overview'&&<><div className="cvx-grid four"><article><small>Companies</small><strong>{companies.length}</strong></article><article><small>Leads loaded</small><strong>{leads.length}</strong></article><article><small>Plans configured</small><strong>{plans.length}</strong></article><article><small>Public pages</small><strong>2</strong></article></div><div className="cvx-panel"><h2>Control Center modules</h2><p>Use the tabs above to configure lead operations, integrations, subscription plan drafts, feature visibility and public page content. Database-backed persistence and secure payment verification are part of the backend completion phase.</p></div></>}
 {tab==='leads'&&<><div className="cvx-toolbar"><select value={selectedCompany} onChange={e=>setSelectedCompany(e.target.value)}><option value="">All companies</option>{companies.map(c=><option key={c.id} value={c.id}>{c.name}</option>)}</select><input placeholder="Search lead, phone or source" value={search} onChange={e=>setSearch(e.target.value)}/></div><div className="cvx-table-wrap"><table><thead><tr><th>Lead</th><th>Source / Received</th><th>Status</th><th>Assigned agent</th><th>Transfer / Assign</th></tr></thead><tbody>{visible.map(l=><tr key={l.id}><td><b>{l.customer_name}</b><small>{l.mobile||'No mobile'}</small></td><td>{l.source||'Unknown'}<small>{new Date(l.created_at).toLocaleString('en-IN')}</small></td><td>{l.status||'New'}</td><td>{profiles.find(p=>p.id===l.assigned_to)?.full_name||'Unassigned'}</td><td><select value={l.assigned_to||''} onChange={e=>void assign(l,e.target.value)}><option value="">Unassigned</option>{profiles.filter(p=>p.company_id===l.company_id).map(p=><option key={p.id} value={p.id}>{p.full_name||p.id.slice(0,8)} · {p.role||'member'}</option>)}</select></td></tr>)}</tbody></table>{visible.length===0&&<p className="cvx-empty">No leads found for this filter.</p>}</div><p className="cvx-help">Assignment uses the existing leads.assigned_to field. Transfer history and response-time event tracking require the backend audit additions in the final backend phase.</p></>}
 {tab==='integrations'&&<div className="cvx-panel"><h2>API Integration Manager</h2><p>Configure provider details in this interface. Do not paste production secrets into this frontend draft: secure credential storage and webhook execution will be connected server-side later.</p><div className="cvx-form-grid"><label>Provider<select value={apiProvider} onChange={e=>setApiProvider(e.target.value)}>{['99acres','Magicbricks','Housing.com','Razorpay','WhatsApp Business','SMS Provider','Calling Provider','Custom Webhook'].map(x=><option>{x}</option>)}</select></label><label>Webhook / API endpoint<input value={apiUrl} onChange={e=>setApiUrl(e.target.value)} placeholder="https://provider.example/webhook"/></label><label>API key placeholder<input type="password" value={apiKey} onChange={e=>setApiKey(e.target.value)} placeholder="Connect secure vault in backend phase"/></label><label className="cvx-check"><input type="checkbox" checked={apiEnabled} onChange={e=>setApiEnabled(e.target.checked)}/> Enable integration (draft)</label></div><button className="cvx-btn" onClick={()=>setNotice(`${apiProvider} configuration is a local frontend draft; it is not connected to the provider yet.`)}>Save configuration draft</button></div>}
 {tab==='plans'&&<><div className="cvx-panel"><h2>Subscription Plans & Payments</h2><p>Manage plan display/pricing drafts, QR and UPI details. Live checkout, payment capture and automatic verification require a payment gateway webhook in the backend.</p><div className="cvx-plan-list">{plans.map((p,i)=><div className="cvx-plan" key={p.name}><label>Plan name<input value={p.name} onChange={e=>setPlans(old=>old.map((x,j)=>i===j?{...x,name:e.target.value}:x))}/></label><label>Price (INR)<input value={p.price} onChange={e=>setPlans(old=>old.map((x,j)=>i===j?{...x,price:e.target.value}:x))}/></label><label>Validity<input value={p.duration} onChange={e=>setPlans(old=>old.map((x,j)=>i===j?{...x,duration:e.target.value}:x))}/></label><label>Description<input value={p.description} onChange={e=>setPlans(old=>old.map((x,j)=>i===j?{...x,description:e.target.value}:x))}/></label><label className="cvx-check"><input type="checkbox" checked={p.enabled} onChange={e=>setPlans(old=>old.map((x,j)=>i===j?{...x,enabled:e.target.checked}:x))}/> Enabled</label></div>)}</div><button className="cvx-btn" onClick={()=>setPlans(old=>[...old,{name:'New Plan',price:'',duration:'Monthly',description:'',enabled:true}])}>+ Add plan</button></div><div className="cvx-panel"><h2>Manual QR / UPI details</h2><div className="cvx-form-grid"><label>UPI ID<input value={upi} onChange={e=>setUpi(e.target.value)} placeholder="business@upi"/></label><label>QR image URL<input value={qr} onChange={e=>setQr(e.target.value)} placeholder="https://.../payment-qr.png"/></label></div><p className="cvx-help">Manual QR payments cannot be auto-verified from a screenshot. Use a payment provider webhook and server-side signature verification for automatic activation.</p><button className="cvx-btn" onClick={()=>setNotice('QR / UPI details saved only in this browser draft. Backend persistence is pending.')}>Save payment draft</button></div></>}
 {tab==='features'&&<div className="cvx-panel"><h2>Feature Manager</h2><p>Feature visibility draft. Actual authorization must be enforced by server-side role checks and database policies.</p><div className="cvx-feature-grid">{defaultFeatures.map(f=><label key={f}><input type="checkbox" defaultChecked/> {f}</label>)}</div></div>}
 {tab==='pages'&&<div className="cvx-two">{(['about','privacy'] as const).map(key=><section className="cvx-panel" key={key}><h2>{key==='about'?'About CODEVEX':'Privacy Policy'}</h2><label>Page title<input value={pages[key].title} onChange={e=>setPages(old=>({...old,[key]:{...old[key],title:e.target.value}}))}/></label><label>Page content<textarea rows={9} value={pages[key].body} onChange={e=>setPages(old=>({...old,[key]:{...old[key],body:e.target.value}}))}/></label><label className="cvx-check"><input type="checkbox" checked={pages[key].published} onChange={e=>setPages(old=>({...old,[key]:{...old[key],published:e.target.checked}}))}/> Publish page (draft visibility only)</label><a href={key==='about'?'/about':'/privacy-policy'} target="_blank" rel="noreferrer">Preview public page ↗</a></section>)}</div>}
 {tab==='audit'&&<div className="cvx-panel"><h2>Audit & Activity</h2><p>Current frontend displays recent lead records and source timestamps in Lead Control. Platform audit log querying, export history, integration events, assignment transfer history and immutable audit records will be connected after the schema audit.</p><button className="cvx-btn" onClick={()=>setTab('leads')}>Open lead activity</button></div>}
 </div>
}
